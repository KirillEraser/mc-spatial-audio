cmake_minimum_required(VERSION 3.24)
set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum supported macOS version")
project(mcspatial VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)

if(BUILD_TESTING)
    add_executable(channel_order_test tests/ChannelOrderTest.cpp)
    target_include_directories(channel_order_test PRIVATE src)
    add_test(NAME channel_order COMMAND channel_order_test)
endif()

option(MCSPATIAL_BUILD_MAC_BACKEND "Build Apple backend and JNI library" ${APPLE})
if(MCSPATIAL_BUILD_MAC_BACKEND)
    if(NOT APPLE)
        message(FATAL_ERROR "The audio backend requires macOS; disable MCSPATIAL_BUILD_MAC_BACKEND for portable tests")
    endif()
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 17)
    set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
    find_package(JNI REQUIRED)
    find_library(AUDIO_TOOLBOX AudioToolbox REQUIRED)
    find_library(CORE_AUDIO CoreAudio REQUIRED)
    find_library(FOUNDATION Foundation REQUIRED)
    find_library(SECURITY Security REQUIRED)
    find_library(APP_KIT AppKit REQUIRED)

    add_library(mcspatial_core STATIC src/SpatialOutput.mm)
    set_target_properties(mcspatial_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_include_directories(mcspatial_core PUBLIC src)
    target_compile_options(mcspatial_core PRIVATE -Wall -Wextra -Wpedantic -fobjc-arc)
    target_link_libraries(mcspatial_core PUBLIC ${AUDIO_TOOLBOX} ${CORE_AUDIO} ${FOUNDATION} ${SECURITY})

    add_library(mcspatial SHARED src/JniBridge.cpp)
    target_include_directories(mcspatial PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(mcspatial PRIVATE mcspatial_core)
    target_compile_options(mcspatial PRIVATE -Wall -Wextra -Wpedantic)
    set_target_properties(mcspatial PROPERTIES OUTPUT_NAME mcspatial)

    add_executable(MCSpatialProbe MACOSX_BUNDLE src/Probe.mm)
    target_link_libraries(MCSpatialProbe PRIVATE mcspatial_core ${APP_KIT})
    target_compile_options(MCSpatialProbe PRIVATE -Wall -Wextra -Wpedantic -fobjc-arc)
    set_target_properties(MCSpatialProbe PROPERTIES
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/probe/Info.plist.in")
endif()
